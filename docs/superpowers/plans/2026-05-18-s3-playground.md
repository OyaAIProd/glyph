# S3 — Playground at `seanhanca.github.io/glyph/play/` (GitHub-native)

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended). Each PR is one subagent dispatch. The master PR cycle template lives in `2026-05-18-tier-s-master.md`.

**Goal:** Ship a public web playground hosted entirely on GitHub Pages, colocated with the source under `site/play/`, that lets users paste a CSV, edit a spec, see chart + audit findings + trust score, and share via URL or Gist. No Vercel, no external KV, no domain purchases. Everything one PR can land or revert.

**Architecture:** Pure static HTML/JS in `site/play/` (next to `site/index.html`). Vanilla ES modules, no bundler. CDN imports for Monaco editor and DuckDB-wasm. `@glyph/core` ships as a pre-bundled ESM file (`glyph-bundle.js`) committed to the repo and regenerated by a release script. GitHub Pages serves it directly from the `gh-pages` branch (built by an Action on every push to `main`). Share storage uses URL hash (no network) for compact specs; for large specs / persistent links, a "Save as Gist" button uses the GitHub Gist API — anonymous gists for instant sharing, or "Save to my account" for the user's signed-in flow.

**Tech stack:**
- HTML + ES modules + plain CSS (matches `site/index.html` style)
- Monaco editor via CDN (`https://cdn.jsdelivr.net/npm/monaco-editor@0.x/min/vs/loader.js`)
- DuckDB-wasm via CDN (`https://cdn.jsdelivr.net/npm/@duckdb/duckdb-wasm@0.x/+esm`)
- `@glyph/core` pre-bundled via `esbuild` to a single ESM file (`site/play/glyph-bundle.js`)
- `pako` (for share-URL compression) via CDN
- GitHub Pages for hosting (free, custom domain support)
- GitHub Gist API for large-spec persistence (no auth required for anonymous gists; rate-limited by client IP)

**Effort:** M (6 PRs, ~2 calendar weeks).

**Calendar:** Weeks 3–5.

**Hosting URL:** `https://seanhanca.github.io/glyph/play/` (free, no domain needed). If a custom domain like `glyph.dev` is later added, just drop a `site/CNAME` file — GitHub Pages picks it up automatically.

```js
// site/play/duckdb.js
// Lazy-loads DuckDB-wasm from jsdelivr, exposes a tiny API.

import * as duckdb from "https://cdn.jsdelivr.net/npm/@duckdb/duckdb-wasm@1.28.0/+esm";

let _db = null;

export async function getDuckDb() {
  if (_db) return _db;
  const bundle = await duckdb.selectBundle(duckdb.getJsDelivrBundles());
  const workerUrl = URL.createObjectURL(
    new Blob([`importScripts("${bundle.mainWorker}");`], { type: "text/javascript" }),
  );
  const worker = new Worker(workerUrl);
  const logger = new duckdb.ConsoleLogger();
  const db = new duckdb.AsyncDuckDB(logger, worker);
  await db.instantiate(bundle.mainModule, bundle.pthreadWorker);
  _db = db;
  return db;
}

export async function loadCsv(tableName, csv) {
  const db = await getDuckDb();
  const safe = tableName.replace(/[^a-zA-Z0-9_]/g, "_");
  await db.registerFileText(`${safe}.csv`, csv);
  const conn = await db.connect();
  try {
    await conn.query(`CREATE OR REPLACE TABLE ${safe} AS SELECT * FROM read_csv_auto('${safe}.csv')`);
  } finally {
    await conn.close();
  }
  return safe;
}

export async function queryRows(sql) {
  const db = await getDuckDb();
  const conn = await db.connect();
  try {
    const result = await conn.query(sql);
    return result.toArray().map((r) => Object.fromEntries(Object.entries(r)));
  } finally {
    await conn.close();
  }
}

export async function describeTable(tableName) {
  return queryRows(`DESCRIBE ${tableName}`);
}
```

```js
// inside playground.js
import { loadCsv, queryRows, describeTable } from "./duckdb.js";

function mountCsvUpload(host, onLoaded) {
  host.innerHTML = `
    <input type="file" id="csv-file" accept=".csv,text/csv" />
    <p class="hint">or paste:</p>
    <textarea id="csv-paste" placeholder="hour,rides\n0,42\n1,38\n..." rows="8"></textarea>
    <p id="csv-status" class="hint"></p>
  `;
  const status = host.querySelector("#csv-status");

  const handle = async (csv) => {
    status.textContent = "Loading…";
    try {
      const table = await loadCsv("data", csv);
      const cols = await describeTable(table);
      const count = (await queryRows(`SELECT COUNT(*) AS n FROM ${table}`))[0].n;
      status.textContent = `Loaded ${count} rows, ${cols.length} columns`;
      onLoaded({ table, rows: await queryRows(`SELECT * FROM ${table}`), columns: cols });
    } catch (e) {
      status.textContent = `Error: ${e.message ?? e}`;
    }
  };

  host.querySelector("#csv-file").addEventListener("change", async (e) => {
    const file = e.target.files?.[0];
    if (file) handle(await file.text());
  });
  host.querySelector("#csv-paste").addEventListener("blur", (e) => {
    if (e.target.value) handle(e.target.value);
  });
}
```

- [ ] **Step 1: Wire Monaco via the AMD loader (it's the official path)**

```js
// site/play/monaco-bootstrap.js
const MONACO_VERSION = "0.45.0";
const MONACO_BASE = `https://cdn.jsdelivr.net/npm/monaco-editor@${MONACO_VERSION}/min/vs`;

let _monacoPromise = null;

export function loadMonaco() {
  if (_monacoPromise) return _monacoPromise;
  _monacoPromise = new Promise((resolve, reject) => {
    const script = document.createElement("script");
    script.src = `${MONACO_BASE}/loader.js`;
    script.onerror = () => reject(new Error("monaco loader failed"));
    script.onload = () => {
      // global `require` is now defined by the AMD loader
      window.require.config({ paths: { vs: MONACO_BASE } });
      window.require(["vs/editor/editor.main"], () => resolve(window.monaco));
    };
    document.head.appendChild(script);
  });
  return _monacoPromise;
}

export async function mountSpecEditor(host, initialValue, onChange) {
  const monaco = await loadMonaco();

  // Load the Glyph JSON schema for autocomplete + validation
  const schema = await fetch("./spec.schema.json").then((r) => r.json());
  monaco.languages.json.jsonDefaults.setDiagnosticsOptions({
    validate: true,
    schemas: [
      {
        uri: "https://glyph.dev/spec.schema.json",
        fileMatch: ["*"],
        schema,
      },
    ],
  });

  const editor = monaco.editor.create(host, {
    value: initialValue,
    language: "json",
    minimap: { enabled: false },
    fontSize: 13,
    scrollBeyondLastLine: false,
    wordWrap: "on",
    automaticLayout: true,
    theme: "vs",
  });

  editor.onDidChangeModelContent(() => {
    onChange(editor.getValue());
  });

  return editor;
}
```

```js
    auditHost.innerHTML = `<li class="error">audit error: ${e.message}</li>`;
    return;
  }
  trustHost.textContent = `${auditResult.trust} / 100`;
  trustHost.style.color = auditResult.trust < 50 ? "var(--red)" : auditResult.trust < 80 ? "#d4a017" : "var(--green)";

  if (auditResult.findings.length === 0) {
    auditHost.innerHTML = `<li class="clean">✓ no findings</li>`;
    return;
  }
  auditHost.innerHTML = auditResult.findings
    .map((f) => `
      <li class="${f.severity}">
        <strong>${escapeHtml(f.rule)}</strong>
        <div>${escapeHtml(f.message)}</div>
        ${f.fix ? `<code>fix: ${escapeHtml(f.fix)}</code>` : ""}
      </li>
    `)
    .join("");
}

function escapeHtml(s) {
  return String(s).replace(/[&<>"']/g, (c) => ({
    "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;",
  }[c]));
}
```

### Share design

**Two tiers:**

1. **URL-hash sharing (no network).** Compressed JSON in the URL fragment. Works offline, no auth, no rate limits. Cap: ~4KB compressed (URL length safe limit ~8KB; budget the rest for the origin + path). Suits any small-data spec.

2. **GitHub Gist sharing (for large data or persistent links).** Two flavors:
   - **Anonymous gist** — POST to `https://api.github.com/gists` with no auth. GitHub allows anonymous gist creation but it's rate-limited per IP (~60/hour). The created gist is public; we get a URL like `gist.github.com/<id>`.
   - **Save to my account** — opens the GitHub create-gist UI pre-filled (`https://gist.github.com/new?filename=playground.glyph.json&description=…`). The user signs in with their own account and clicks "Create". Their gist, their account, no API limits for us. We then read the resulting URL from a paste/return flow.

**Reading shared playgrounds** — on load, the playground inspects the URL:
- `#h=<base64>` → URL-hash mode; decode and load
- `?gist=<id>` → fetch `https://gist.githubusercontent.com/raw/<id>` (or via the Gists API to get all files)

```js
// site/play/share.js
import * as pako from "https://cdn.jsdelivr.net/npm/pako@2.1.0/+esm";

// ---------- URL hash mode ----------

export function encodeHash(payload) {
  const json = JSON.stringify(payload);
  const compressed = pako.deflate(new TextEncoder().encode(json));
  return base64UrlEncode(compressed);
}

export function decodeHash(hash) {
  const bytes = base64UrlDecode(hash);
  const json = pako.inflate(bytes, { to: "string" });
  return JSON.parse(json);
}

// ---------- Anonymous Gist mode ----------

export async function createAnonymousGist({ spec, csv, description }) {
  const files = {
    "playground.glyph.json": { content: JSON.stringify(spec, null, 2) },
  };
  if (csv) files["data.csv"] = { content: csv };
  const res = await fetch("https://api.github.com/gists", {
    method: "POST",
    headers: { "content-type": "application/json", accept: "application/vnd.github+json" },
    body: JSON.stringify({
      description: description || "Glyph playground",
      public: true,
      files,
    }),
  });
  if (!res.ok) {
    if (res.status === 403) {
      throw new Error("GitHub rate limit reached. Try the 'Save to my account' button instead.");
    }
    throw new Error(`Gist API error: ${res.status} ${res.statusText}`);
  }
  const data = await res.json();
  return { id: data.id, html_url: data.html_url };
}

export async function fetchGist(id) {
  const res = await fetch(`https://api.github.com/gists/${id}`, {
    headers: { accept: "application/vnd.github+json" },
  });
  if (!res.ok) throw new Error(`Gist not found: ${id}`);
  const data = await res.json();
  const specFile = data.files["playground.glyph.json"];
  const csvFile = data.files["data.csv"];
  return {
    spec: specFile ? JSON.parse(specFile.content) : null,
    csv: csvFile ? csvFile.content : null,
  };
}

// ---------- "Save to my account" — opens GitHub UI ----------

export function openSaveToMyAccount({ spec, csv }) {
  // Use the create-gist URL with prefilled content. The user signs in to GH
  // and clicks "Create gist" — their gist, their account, no rate limits for us.
  const params = new URLSearchParams();
  params.set("filename", "playground.glyph.json");
  params.set("description", "Glyph playground");
  // GitHub's create-gist UI doesn't officially support `?content=` prefill —
  // but it does accept `value` via the form. As a workaround, copy the content
  // to clipboard and open the blank gist creator.
  const block = `${JSON.stringify(spec, null, 2)}\n\n# data.csv\n${csv ?? ""}`;
  navigator.clipboard.writeText(block);
  alert("Content copied to clipboard. Paste it into the gist that opens next.");
  window.open("https://gist.github.com/", "_blank");
}

// ---------- helpers ----------

function base64UrlEncode(bytes) {
  let bin = "";
  for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
  return btoa(bin).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
}

function base64UrlDecode(s) {
  const padded = s.replace(/-/g, "+").replace(/_/g, "/") +
    "=".repeat((4 - (s.length % 4)) % 4);
  const bin = atob(padded);
  const bytes = new Uint8Array(bin.length);
  for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
  return bytes;
}
```

```js
function toCsv(rows, columns) {
  const header = columns.join(",");
  const body = rows.map((r) =>
    columns.map((c) => {
      const v = r[c];
      if (v === null || v === undefined) return "";
      const s = String(v);
      return /[,"\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;
    }).join(","),
  ).join("\n");
  return `${header}\n${body}\n`;
}
```

## Why this is better than the Vercel/KV version

| Concern | Vercel/KV plan | GitHub-only plan |
|---|---|---|
| Account needed | Vercel + Vercel KV + GitHub | GitHub only |
| Monthly cost ceiling | Free tier → quotas → paid | Free, no quotas relevant at this scale |
| Source-code distance | Separate repo or `apps/playground` | `site/play/` next to landing page |
| Bundler | Next.js + Tailwind + TS | None — vanilla ESM, viewable source |
| Share durability | Tied to Vercel KV plan | Gists are forever; URLs work indefinitely |
| User ownership | Glyph owns all share data | "Save to my account" → user owns their gist |
| Discoverability | Custom DB to surface gallery | GitHub Discussions, free, indexed by Google |
| Maintenance ops | Watch KV usage, manage env vars | None |
| Migration if Glyph dies | Hard | None — gists outlive the repo |
